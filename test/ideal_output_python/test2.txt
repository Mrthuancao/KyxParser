            event_collision.direction = -1 # Trigger when value goes from positive to negative (i.e. distance becomes 0 or negative)

        self.reward_range = (-float('inf'), float('inf'))

    def _compile_ode_func(self, func_str):
        exec_globals = {'np': np, 'math': math}
        exec_locals = {'self': self}
        exec(textwrap.dedent(func_str), exec_globals, exec_locals)
        return exec_locals['ode_func']

    def _compile_ode_events(self, events_str):
        if not events_str.strip():
            return []
        exec_globals = {'np': np, 'math': math}
        exec_locals = {'self': self}
        exec(textwrap.dedent(events_str), exec_globals, exec_locals)
        return exec_locals['ode_events']

    def _get_obs(self):
        return np.array([self.state[key] for key in self._state_keys], dtype=np.float32)

    def _get_info(self):
        return self.state.copy()

    def _evaluate_expression(self, expr_str, local_vars=None):
        if local_vars is None:
            local_vars = {}
        if expr_str.startswith('?'):
            expr_str = expr_str[1:].strip()

        eval_globals = {
            'np': np, 'math': math,
            **self.constants,
        }
        # Ensure functions and predicates can access constants and other functions/predicates
        for k, v in self.functions.items():
            eval_globals[k] = v
        for k, v in self.predicates.items():
            eval_globals[k] = v

        eval_locals = {
            **self.state,
            **local_vars
        }

        expr_py = expr_str.replace('^', '**').replace('abs(', 'np.abs(')
        expr_py = expr_py.replace('&', 'and').replace('|', 'or')
        expr_py = re.sub(r'(?<![<>=!])=(?![=])', '==', expr_py)
        
        # FIX: Replace '->' (implication) with its logical equivalent: 'not A or B'
        # This regex looks for 'A -> B' where A and B are expressions, and replaces it.
        # It's a simplified regex and might not handle all complex nested implications perfectly,
        # but it should work for the current format `(condition -> result)`.
        # It assumes the `->` is within parentheses or clearly separates two logical parts.
        # For `(v>0 -> (x - xo)**2 + (y - yo)**2 > 0)`, it will match `v>0` as group 1 and `(x - xo)**2 + (y - yo)**2 > 0` as group 2.
        expr_py = re.sub(r'\(?\s*([^)]+?)\s*->\s*([^)]+?)\s*\)?', r'(not (\1) or (\2))', expr_py)


        try:
            return eval(expr_py, eval_globals, eval_locals)
        except Exception as e:
            print(f"Error evaluating expression '{expr_str}': {e}")
            print(f"Expression: '{expr_str}'")
            print(f"Pythonized: '{expr_py}'")
            print(f"Eval globals: {eval_globals}")
            print(f"Eval locals: {eval_locals}")
            raise

    def _process_action_config_node(self, node_config, action, reward, terminated, control_applied):
        """
        Recursively processes an action configuration node and applies its effects.
        Returns updated reward, terminated status, and control_applied status.
        """
        if terminated or not control_applied:
            return reward, terminated, control_applied

        if isinstance(node_config, SequenceActionConfig):
            for sub_action_config in node_config.actions:
                reward, terminated, control_applied = self._process_action_config_node(sub_action_config, action, reward, terminated, control_applied)
                if terminated or not control_applied:
                    return reward, terminated, control_applied
            return reward, terminated, control_applied
        
        elif isinstance(node_config, ActionConstraintConfig):
            constraint_met = self._evaluate_expression(node_config.formula)
            if not constraint_met:
                # print(f"Constraint violated: {node_config.formula}") # Debug print
                reward -= 100.0
                terminated = True
                control_applied = False
            return reward, terminated, control_applied
        
        elif isinstance(node_config, DeterministicAssignmentActionConfig):
            assigned_value = self._evaluate_expression(node_config.value)
            self.state[node_config.variable] = assigned_value
            return reward, terminated, control_applied
        
        elif isinstance(node_config, NondeterministicAssignmentActionConfig):
            assign_key = f"assign_{node_config.variable}"
            if assign_key in action:
                self.state[node_config.variable] = action[assign_key]
            else:
                # This should not happen if action space is constructed correctly and agent provides all values
                # print(f"Warning: Nondeterministic assignment for {node_config.variable} not found in action.")
                reward -= 50.0
                terminated = True
                control_applied = False
            return reward, terminated, control_applied
        
        elif isinstance(node_config, ChoiceActionConfig):
            choice_key = f"choice_{node_config.level_name}"
            if choice_key not in action:
                # Agent did not provide a choice for this level
                # print(f"Warning: Choice key '{choice_key}' not found in action.")
                reward -= 50.0
                terminated = True
                control_applied = False
                return reward, terminated, control_applied

            choice_idx = action[choice_key]
            if not (0 <= choice_idx < len(node_config.choices)):
                # Invalid choice index
                # print(f"Warning: Invalid choice index {choice_idx} for {choice_key}.")
                reward -= 50.0
                terminated = True
                control_applied = False
                return reward, terminated, control_applied

            chosen_sub_config = node_config.choices[choice_idx]
            return self._process_action_config_node(chosen_sub_config, action, reward, terminated, control_applied)
        
        else:
            print(f"Warning: Unknown action config type encountered: {type(node_config)}")
            reward -= 10.0
            control_applied = False
            return reward, terminated, control_applied

    def _apply_control_action(self, action):
        reward = 0
        terminated = False
        control_applied = True # Assume success unless constraint violated or assignment missing

        # Store previous state for reward calculation (control effort)
        prev_v, prev_w = self.state['v'], self.state['w']

        # Start processing from the top-level control function configuration
        reward, terminated, control_applied = self._process_action_config_node(
            self._control_func_action_config, action, reward, terminated, control_applied
        )

        if not control_applied: # If any issue during processing, return early
            return reward, terminated, control_applied
        
        # FIX: Update 'r' based on 'v' and 'w' after control actions
        if self.state['w'] != 0:
            self.state['r'] = self.state['v'] / self.state['w']
        else:
            self.state['r'] = np.inf # If w is 0, radius is infinite (straight line)

        self.state['t'] = 0.0 # Reset local time for continuous evolution

        # Calculate control effort penalty
        reward -= 0.05 * ((self.state['v'] - prev_v)**2 + (self.state['w'] - prev_w)**2)

        return reward, terminated, control_applied

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self.current_step = 0

        # --- Initial State (Randomized for Robust Learning) ---
        self.state['x'] = self.np_random.uniform(-2.0, 2.0)
        self.state['y'] = self.np_random.uniform(-2.0, 2.0)
        self.state['v'] = self.np_random.uniform(0.5, 2.0) # Start with some velocity
        self.state['a'] = 0.0
        
        # Randomize initial direction, ensuring it's a unit vector
        angle = self.np_random.uniform(0, 2 * np.pi)
        self.state['dx'] = np.cos(angle)
        self.state['dy'] = np.sin(angle)
        
        self.state['w'] = self.np_random.uniform(-0.5, 0.5) # Initial angular velocity
        if self.state['w'] != 0:
            self.state['r'] = self.state['v'] / self.state['w']
        else:
            self.state['r'] = np.inf

        # Obstacle initial position and velocity
        # Ensure obstacle is initially separated from robot
        min_initial_dist = 3.0
        while True:
            self.state['xo'] = self.np_random.uniform(-7.0, 7.0)
            self.state['yo'] = self.np_random.uniform(-7.0, 7.0)
            dist_sq = (self.state['x'] - self.state['xo'])**2 + (self.state['y'] - self.state['yo'])**2
            if dist_sq > min_initial_dist**2:
                break

        self.state['vxo'] = self.np_random.uniform(-self.constants['V'] * 0.5, self.constants['V'] * 0.5)
        self.state['vyo'] = self.np_random.uniform(-self.constants['V'] * 0.5, self.constants['V'] * 0.5)
        # Ensure obstacle velocity is within bounds
        if self.state['vxo']**2 + self.state['vyo']**2 > self.constants['V']**2:
            norm = np.sqrt(self.state['vxo']**2 + self.state['vyo']**2)
            self.state['vxo'] = self.state['vxo'] / norm * self.constants['V'] * 0.5 # Scale down to half max V
            self.state['vyo'] = self.state['vyo'] / norm * self.constants['V'] * 0.5

        self.state['t'] = 0.0

        # Optional: Validate initial state against assumptions (for debugging)
        # if not self._evaluate_expression('assumptions(x,y,v,dx,dy,xo,yo)', self.state):
        #     print("Warning: Initial state does not satisfy assumptions!")

        observation = self._get_obs()
        info = self._get_info()
        return observation, info

    def step(self, action):
        self.current_step += 1
        reward = 0.0
        terminated = False
        truncated = False
        info = self._get_info()

        # Store previous state for reward shaping
        prev_robot_pos = np.array([self.state['x'], self.state['y']])
        prev_obstacle_pos = np.array([self.state['xo'], self.state['yo']])
        prev_dist_sq = np.sum((prev_robot_pos - prev_obstacle_pos)**2)

        # --- 1. Apply Control Function (Discrete Action) ---
        control_reward, control_terminated, control_applied = self._apply_control_action(action)
        reward += control_reward
        terminated = terminated or control_terminated

        if not control_applied:
            return self._get_obs(), reward, terminated, truncated, info

        # --- 2. Simulate Differential Equations (Continuous Dynamics) ---
        y0 = np.array([self.state[var_name] for var_name in self._ode_state_map.keys()], dtype=np.float64)
        t_span = (0, self.constants['ep'])

        try:
            solution = solve_ivp(
                self._ode_func_compiled, 
                t_span, 
                y0, 
                events=self._ode_events_compiled, 
                dense_output=True,
                rtol=1e-6, atol=1e-8
            )

            if solution.status == 0: # Integration completed successfully
                final_y = solution.y[:, -1]
                for i, var_name in enumerate(self._ode_state_map.keys()):
                    self.state[var_name] = final_y[i]
            elif solution.status == 1: # An event occurred
                # Find the earliest event time if multiple events trigger at the same time
                event_indices = [i for i, te in enumerate(solution.t_events) if te.size > 0]
                
                if event_indices:
                    # Get the time of the first event that occurred
                    event_times = [solution.t_events[idx][0] for idx in event_indices]
                    earliest_event_time = min(event_times)
                    
                    # Get the state at that event time using dense_output
                    final_y = solution.sol(earliest_event_time)
                else:
                    # Fallback if event data is missing or misinterpreted, take last state
                    final_y = solution.y[:, -1]

                for i, var_name in enumerate(self._ode_state_map.keys()):
                    self.state[var_name] = final_y[i]

                # Now check which type of event terminated it and apply penalties
                # Event 0: t_max (time limit reached, usually not a penalty unless it's a failure to reach goal)
                # Event 1: v_min (v < 0, violation)
                # Event 2: collision (distance <= 0, violation)
                
                # Re-evaluate conditions at the final state to determine specific termination cause
                if self.state['v'] < 0:
                    reward -= 500.0 # Severe penalty for v < 0
                    terminated = True
                
                current_dist_sq = (self.state['x'] - self.state['xo'])**2 + (self.state['y'] - self.state['yo'])**2
                if current_dist_sq <= 1e-6: # Treat very small distance as collision
                    reward -= 1000.0 # Very severe penalty for collision
                    terminated = True
                
            else: # Integration failed (status 2)
                # print(f"ODE integration failed: {solution.message}")
                reward -= 200.0
                terminated = True

        except Exception as e:
            # print(f"Error during ODE integration: {e}")
            reward -= 300.0
            terminated = True
        
        # Ensure dx, dy remain a unit vector after ODE step (orientation)
        norm_dir = np.sqrt(self.state['dx']**2 + self.state['dy']**2)
        if norm_dir > 1e-6:
            self.state['dx'] /= norm_dir
            self.state['dy'] /= norm_dir
        else: # Fallback if direction vector collapses (should be rare with w!=0 or v=0)
            self.state['dx'] = 1.0
            self.state['dy'] = 0.0
            reward -= 50.0 # Penalty for direction vector issue

        # Update 'r' after ODE step as well, as v and w might have changed (though they are constant in ODEs,
        # self.state['r'] needs to reflect the current derived value for subsequent evaluations).
        if self.state['w'] != 0:
            self.state['r'] = self.state['v'] / self.state['w']
        else:
            self.state['r'] = np.inf

        # --- 3. Reward Shaping ---
        current_robot_pos = np.array([self.state['x'], self.state['y']])
        current_obstacle_pos = np.array([self.state['xo'], self.state['yo']])
        current_dist_sq = np.sum((current_robot_pos - current_obstacle_pos)**2)
        current_dist = np.sqrt(current_dist_sq)

        # Reward for staying safe (positive reward per step)
        reward += 1.0

        # Reward/penalty based on distance to admissible separation
        admissible_sep = self._evaluate_expression('admissibleSeparation(v)', {'v': self.state['v']})
        
        if current_dist > admissible_sep:
            reward += 0.5 * (current_dist - admissible_sep) # Reward for being further away than necessary
        else:
            # Penalty for being too close, but not yet collided
            reward -= 2.0 * (admissible_sep - current_dist)
            # Add a stronger gradient penalty if very close to collision
            if current_dist < 0.5 * admissible_sep:
                 reward -= 5.0 * (0.5 * admissible_sep - current_dist)

        # Reward for making progress (e.g., maintaining speed)
        reward += 0.1 * self.state['v']

        # --- 4. Check Problem Postcondition (Safety) ---
        # This is already handled by the collision event and v<0 check.
        # But let's keep the explicit check for consistency and additional penalty if not caught by event.
        postcondition_satisfied = self._evaluate_expression('(v>0 -> (x - xo)**2 + (y - yo)**2 > 0)')
        if not postcondition_satisfied and not terminated: # Only apply if not already terminated by event
            reward -= 500.0 # Safety violation
            terminated = True

        # --- 5. Check Truncation ---
        if self.current_step >= self.max_episode_steps:
            truncated = True
            reward -= 50.0 # Penalty for truncating without reaching goal/long survival

        observation = self._get_obs()
        info = self._get_info()
        return observation, reward, terminated, truncated, info

            print(f"Render: Step={self.current_step}, State={self.state}")
