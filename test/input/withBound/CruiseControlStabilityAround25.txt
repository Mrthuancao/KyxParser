Theorem "Automatic cruise controller: stability"
/*
 * Generated from hybrid automaton
 * subgraph CruiseController
 * ebrakeact("v' = -t-2.5, x' = 0, t' = 1
 *       & 15 <= v & v <= 40
 *       & 0 <= t & t <= 2.5")
 * ebrakefull("v' = -5, x' = 0, t' = 0
 *       & 15 <= v & v <= 40
 *       & t = 2.5")
 * sbrakeact("v' = -t-1.2, x' = 0, t' = 0.5
 *       & 5 <= v & v <= 20
 *       & 0 <= t & t <= 1.3")
 * sbrakefull("v' = -2.5, x' = 0, t' = 0
 *       & 5 <= v & v <= 20
 *       & t = 1.3")
 * normalPI("v' = -0.001*x-0.052*v, x' = v, t' = 0
 *       & -15 <= v & v <= 15
 *       & -500 <= x & x <= 500")
 * accelerate("v' = 1.5, x' = 0, t' = 0
 *       & -20 <= v & v <= -5 ")
 *
 * ebrakeact -->|"?(15 <= v & v <= 40 & t = 2.5);"| ebrakefull
 * ebrakeact -->|"?(15 <= v & v <= 16 & 0 <= t & t <= 2.5); t:=0;"| sbrakeact
 *
 * ebrakefull -->|"?(15 <= v & v <= 16 & t = 2.5); t:=0;"| sbrakeact
 *
 * sbrakeact -->|"?(18 <= v & v <= 20 & 0 <= t & t <= 1.3); t:=0;"| ebrakeact
 * sbrakeact -->|"?(5 <= v & v <= 20 & t = 1.3);"| sbrakefull
 * sbrakeact -->|"?(5 <= v & v <= 11 & 0 <= t & t <= 1.3); x:=0;"| normalPI
 *
 * sbrakefull -->|"?(18 <= v & v <= 20 & t = 1.3); t:=0;"| ebrakeact
 * sbrakefull -->|"?(5 <= v & v <= 11 & t = 1.3); x:=0;"| normalPI
 *
 * normalPI -->|"?(13 <= v & v <= 15 & -500 <= x & x <= 500); t:=0;"| sbrakeact
 * normalPI -->|"?(-15 <= v & v <= -14 & -500 <= x & x <= 500);"| accelerate
 *
 * accelerate -->|"?(-6 <= v & v <= -5 & -500 <= x & x <= 500); x:=0;"| normalPI
 *
 * end
 *
 * Init("t:=0; x:=0;") --> CruiseController
 */

Definitions
  Real accelerate = 0;
  Real ebrakeact = 1;
  Real ebrakefull = 2;
  Real normalPI = 3;
  Real sbrakeact = 4;
  Real sbrakefull = 5;

  // @safety: penalty=-100, on_failure=truncate:status=failure, on_success=continue:status=none
  Bool cruiseSafe() <-> -500 <= x & x <= 500 & 5 <= v & v <= 65 & 0 <= t & t <= 2.5 & 0 <= mode & mode <= 5;

  // @progress: reward=10, shape={-abs(v - 25)}, on_success=continue:status=none, on_failure=continue:status=none
  Bool nearTargetSpeed() <-> abs(v - 25) <= 1;

  HP initializeMode ::= {
    mode:=ebrakeact()
    ++ mode:=ebrakefull()
    ++ mode:=sbrakeact()
    ++ mode:=sbrakefull()
    ++ mode:=normalPI()
    ++ mode:=accelerate()
  };

  HP initialize ::= {
    initializeMode;
    t:=0;
    x:=0
  };

  HP controlFunc ::= {
    /* ebrakeact: Shifted 15->40, 40->65, 16->41 */
    ?mode = ebrakeact(); ?40 <= v & v <= 65 & t = 2.5; mode:=ebrakefull()
    ++ ?mode = ebrakeact(); ?40 <= v & v <= 41 & 0 <= t & t <= 2.5; t:=0; mode:=sbrakeact()
    ++ ?mode = ebrakeact(); ?t < 2.5 & v > 40; mode:=mode

    /* ebrakefull: Shifted 15->40, 16->41 */
    ++ ?mode = ebrakefull(); ?40 <= v & v <= 41 & t = 2.5; t:=0; mode:=sbrakeact()
    ++ ?mode = ebrakefull(); ?v > 40; mode:=mode 

    /* sbrakeact: Shifted 18->43, 20->45, 5->30, 11->36 */
    ++ ?mode = sbrakeact(); ?43 <= v & v <= 45 & 0 <= t & t <= 1.3; t:=0; mode:=ebrakeact()
    ++ ?mode = sbrakeact(); ?30 <= v & v <= 45 & t = 1.3; mode:=sbrakefull()
    ++ ?mode = sbrakeact(); ?30 <= v & v <= 36 & 0 <= t & t <= 1.3; x:=0; mode:=normalPI()
    ++ ?mode = sbrakeact(); ?t < 1.3 & v > 30; mode:=mode

    /* sbrakefull: Shifted 18->43, 20->45, 5->30, 11->36 */
    ++ ?mode = sbrakefull(); ?43 <= v & v <= 45 & t = 1.3; t:=0; mode:=ebrakeact()
    ++ ?mode = sbrakefull(); ?30 <= v & v <= 36 & t = 1.3; x:=0; mode:=normalPI()
    ++ ?mode = sbrakefull(); ?v > 30; mode:=mode

    /* normalPI: Shifted 13->38, 15->40, -15->10, -14->11 */
    ++ ?mode = normalPI(); ?38 <= v & v <= 40 & -500 <= x & x <= 500; t:=0; mode:=sbrakeact()
    ++ ?mode = normalPI(); ?10 <= v & v <= 11 & -500 <= x & x <= 500; mode:=accelerate()
    ++ ?mode = normalPI(); ?v > 10 & v < 40; mode:=mode

    /* accelerate: Shifted -6->19, -5->20 */
    ++ ?mode = accelerate(); ?19 <= v & v <= 20 & -500 <= x & x <= 500; x:=0; mode:=normalPI()
    ++ ?mode = accelerate(); ?v < 20; mode:=mode
  };

  HP continuousDynamics ::= {
    ?mode = ebrakeact(); {v'=-t - 2.5, x'=0, t'=1 & 40 <= v & v <= 65 & 0 <= t & t <= 2.5}
    ++ ?mode = ebrakefull(); {v'=-5, x'=0, t'=0 & 40 <= v & v <= 65 & t = 2.5}
    ++ ?mode = sbrakeact(); {v'=-t - 1.2, x'=0, t'=0.5 & 30 <= v & v <= 45 & 0 <= t & t <= 1.3}
    ++ ?mode = sbrakefull(); {v'=-2.5, x'=0, t'=0 & 30 <= v & v <= 45 & t = 1.3}
    ++ ?mode = normalPI(); {v'=-0.001*x - 0.052*(v - 25), x'=v-25, t'=0 & 10 <= v & v <= 40 & -500 <= x & x <= 500}    
    ++ ?mode = accelerate(); {v'=1.5, x'=0, t'=0 & 5 <= v & v <= 20}
  };

  HP transitionAndFlow ::= {
    controlFunc;
    continuousDynamics
  };

  HP cruiseLoop ::= { transitionAndFlow; }*;

  HP cruiseControl ::= {
    initialize;
    cruiseLoop
  };
End.

ProgramVariables
  // @var: bound=[-500:500], plot=line, def="position error"
  Real x;
  // @var: bound=[0:2.5], plot=line, def="elapsed time in the active braking phase"
  Real t;
  // @var: bound=[5:65], plot=line, def="cruise-control velocity"
  Real v;
  // @var: bound=[0:5], plot=hist, def="active controller mode"
  Real mode;
End.

Problem
  forall(eps).(
    eps > 0 ->
    exists(del).(
      del > 0 &
      forall(v).(
        /* Stability check: If we start within del of 25, we stay within eps of 25 */
        (v - 25)^2 < del^2 ->
        [cruiseControl] (v - 25)^2 < eps^2
      )
    )
  )
End.
