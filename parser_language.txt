/* controller */
HP controller ::= {a:=-b; ++ ?v^2<=2*b*(m-x-ep*v); a:=0;};

/* step */
HP stepFunction ::= {x'=v, v'=a, t'=1 & v>=0 & t<=ep};

/* done */
Bool doneCondition(Real x, Real m) <-> x <= m;
